    MailingState VARCHAR(20) NULL,
    MailingZip CHAR(5) NULL,
    MailingZipPlus4 CHAR(4) NULL,
    MailingCountry VARCHAR(50) NULL,
    MailingPostalCode VARCHAR(10) NULL,

    -- District & School Info
    CareerCenter VARCHAR(80) NULL,
    City VARCHAR(80) NULL,
    CitySchoolDistrict VARCHAR(80) NULL,
    CountyCourtDistrict VARCHAR(80) NULL,
    CongressionalDistrict CHAR(2) NULL,
    CourtOfAppeals CHAR(2) NULL,
    EducationServiceCenter VARCHAR(80) NULL,
    ExemptedVillageSchoolDistrict VARCHAR(80) NULL,
    LibraryDistrict VARCHAR(80) NULL,
    LocalSchoolDistrict VARCHAR(80) NULL,
    MunicipalCourtDistrict VARCHAR(80) NULL,
    Precinct VARCHAR(80) NULL,
    PrecinctCode VARCHAR(20) NULL,
    StateBoardOfEducation CHAR(2) NULL,
    StateRepresentativeDistrict CHAR(2) NULL,
    StateSenateDistrict CHAR(2) NULL,
    Township VARCHAR(30) NULL,
    Village VARCHAR(30) NULL,

-- Step 3: Cursor to Loop Through Each Column in OHIO Table
DECLARE @OhioColumn NVARCHAR(128),
        @SourceColumn NVARCHAR(128),
        @ColumnType NVARCHAR(50), 
        @MaxLength INT, 
        @SQL NVARCHAR(MAX);

DECLARE cur 
	CURSOR FOR 
		SELECT 
			c.COLUMN_NAME AS OhioColumn,
			cm.SourceColumn,
			c.DATA_TYPE + 
			CASE 
				WHEN c.CHARACTER_MAXIMUM_LENGTH IS NOT NULL THEN '(' + CAST(c.CHARACTER_MAXIMUM_LENGTH AS NVARCHAR) + ')'
				ELSE ''
			END AS FieldType,
			ISNULL(c.CHARACTER_MAXIMUM_LENGTH, 0) AS [CHARACTER_MAXIMUM_LENGTH]
		FROM INFORMATION_SCHEMA.COLUMNS c
		JOIN #ColumnMapping cm ON c.COLUMN_NAME = cm.OhioColumn
		WHERE c.TABLE_NAME = 'OHIO';

OPEN cur;
FETCH NEXT FROM cur INTO @OhioColumn, @SourceColumn, @ColumnType, @MaxLength;
WHILE @@FETCH_STATUS = 0
BEGIN
    -- Step 4: Build Dynamic SQL to Check Max Length in Source Data (Using Correct Column Name)
    SET @SQL = '
INSERT INTO #TempResults (FieldName, FieldType, FieldMaxLength, MaxLength, ExampleData)
SELECT TOP 1 ''' + @OhioColumn + ''', ''' + @ColumnType + ''', ''' + CONVERT(NVARCHAR, @MaxLength) + ''', LEN([' + @SourceColumn + ']), [' + @SourceColumn + ']
FROM (
    SELECT ' + @SourceColumn + ' FROM [SWVF_67_88.txt]
    UNION ALL
    SELECT ' + @SourceColumn + ' FROM [SWVF_23_44.txt]
    UNION ALL
    SELECT ' + @SourceColumn + ' FROM [SWVF_1_22.txt]
    UNION ALL
    SELECT ' + @SourceColumn + ' FROM [SWVF_45_66.txt]
) AS SourceData
ORDER BY LEN([' + @SourceColumn + ']) DESC;';

    -- Step 5: Execute the Dynamic SQL
	--PRINT @SQL
    EXEC sp_executesql @SQL;

    FETCH NEXT FROM cur INTO @OhioColumn, @SourceColumn, @ColumnType, @MaxLength;
END
CLOSE cur;
DEALLOCATE cur;
